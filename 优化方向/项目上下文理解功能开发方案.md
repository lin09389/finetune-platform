# 🧠 项目上下文理解功能 - 完整开发方案

## 🎯 功能说明

### 什么是项目上下文理解

```
传统 AI：
用户："帮我优化这段代码"
AI: 给出通用建议（不了解你的项目）

有项目上下文的 AI：
用户："帮我优化这段代码"
AI 知道：
  ✅ 你用的是 React + TypeScript
  ✅ 项目用 Ant Design UI
  ✅ 代码风格是函数式
  ✅ 正在做电商项目
  ✅ 相关组件和 API
AI: 给出符合你项目的精准建议
```

### 第 1 步：创建项目扫描器

```python
# server/context/project_scanner.py

from pathlib import Path
from typing import Dict, List, Any
import json
import re

class ProjectScanner:
    """项目扫描器 - 分析项目结构和特征"""
    
    def __init__(self, project_path: str):
        self.project_path = Path(project_path)
        self.tech_stacks = {
            'python': {
                'files': ['requirements.txt', 'setup.py', 'pyproject.toml'],
                'extensions': ['.py'],
                'ignore': ['__pycache__', '*.pyc', '.git', 'venv', '.venv']
            },
            'javascript': {
                'files': ['package.json', 'tsconfig.json', 'vite.config.ts'],
                'extensions': ['.js', '.ts', '.jsx', '.tsx'],
                'ignore': ['node_modules', 'dist', 'build', '.git']
            },
            'java': {
                'files': ['pom.xml', 'build.gradle'],
                'extensions': ['.java'],
                'ignore': ['target', 'build', '.class']
            }
        }
    
    def scan(self) -> Dict[str, Any]:
        """扫描整个项目"""
        return {
            'name': self.project_path.name,
            'path': str(self.project_path),
            'tech_stack': self._detect_tech_stack(),
            'structure': self._build_structure(),
            'dependencies': self._parse_dependencies(),
            'code_style': self._analyze_code_style(),
            'key_files': self._find_key_files(),
            'git_info': self._get_git_info()
        }
    
    def _detect_tech_stack(self) -> List[str]:
        """检测技术栈"""
        stacks = []
        
        for stack, config in self.tech_stacks.items():
            for file in config['files']:
                if (self.project_path / file).exists():
                    stacks.append(stack)
                    
                    # 检测具体框架
                    if stack == 'javascript':
                        stacks.extend(self._detect_js_frameworks())
                    elif stack == 'python':
                        stacks.extend(self._detect_python_frameworks())
                    break
        
        return stacks
    
    def _detect_js_frameworks(self) -> List[str]:
        """检测 JavaScript 框架"""
        frameworks = []
        pkg_file = self.project_path / 'package.json'
        
        if pkg_file.exists():
            with open(pkg_file, 'r', encoding='utf-8') as f:
                pkg = json.load(f)
                deps = {**pkg.get('dependencies', {}), 
                       **pkg.get('devDependencies', {})}
                
                if 'react' in deps: frameworks.append('React')
                if 'vue' in deps: frameworks.append('Vue')
                if 'angular' in deps: frameworks.append('Angular')
                if 'next' in deps: frameworks.append('Next.js')
                if 'nuxt' in deps: frameworks.append('Nuxt.js')
                if 'antd' in deps: frameworks.append('Ant Design')
                if 'tailwindcss' in deps: frameworks.append('Tailwind CSS')
        
        return frameworks
    
    def _detect_python_frameworks(self) -> List[str]:
        """检测 Python 框架"""
        frameworks = []
        req_file = self.project_path / 'requirements.txt'
        
        if req_file.exists():
            with open(req_file, 'r', encoding='utf-8') as f:
                reqs = f.read().lower()
                
                if 'fastapi' in reqs: frameworks.append('FastAPI')
                if 'flask' in reqs: frameworks.append('Flask')
                if 'django' in reqs: frameworks.append('Django')
                if 'torch' in reqs: frameworks.append('PyTorch')
                if 'tensorflow' in reqs: frameworks.append('TensorFlow')
        
        return frameworks
    
    def _build_structure(self, depth: int = 3) -> Dict:
        """构建项目结构树"""
        def scan_dir(path: Path, current_depth: int) -> Dict:
            if current_depth > depth:
                return {'name': path.name, 'type': 'folder', 'children': []}
            
            structure = {'name': path.name, 'type': 'folder', 'children': []}
            
            try:
                for item in sorted(path.iterdir()):
                    # 跳过忽略的目录
                    if any(ign in str(item) for ign in 
                          ['__pycache__', 'node_modules', '.git', 'venv']):
                        continue
                    
                    if item.is_file():
                        structure['children'].append({
                            'name': item.name,
                            'type': 'file',
                            'size': item.stat().st_size,
                            'path': str(item.relative_to(self.project_path))
                        })
                    else:
                        child = scan_dir(item, current_depth + 1)
                        if child['children']:  # 只添加有内容的目录
                            structure['children'].append(child)
            except PermissionError:
                pass
            
            return structure
        
        return scan_dir(self.project_path, 0)
    
    def _parse_dependencies(self) -> Dict:
        """解析项目依赖"""
        deps = {}
        
        # Python
        req_file = self.project_path / 'requirements.txt'
        if req_file.exists():
            with open(req_file, 'r', encoding='utf-8') as f:
                deps['python'] = [
                    line.strip().split('==')[0]
                    for line in f
                    if line.strip() and not line.startswith('#')
                ]
        
        # JavaScript
        pkg_file = self.project_path / 'package.json'
        if pkg_file.exists():
            with open(pkg_file, 'r', encoding='utf-8') as f:
                pkg = json.load(f)
                deps['javascript'] = {
                    'dependencies': pkg.get('dependencies', {}),
                    'devDependencies': pkg.get('devDependencies', {})
                }
        
        return deps
    
    def _analyze_code_style(self) -> Dict:
        """分析代码风格"""
        style = {
            'indentation': 'space',
            'quote': 'single',
            'naming': 'camelCase',
            'line_length': 88
        }
        
        # 分析 Python 文件
        py_files = list(self.project_path.glob('**/*.py'))[:10]
        if py_files:
            import re
            indent_counts = {'space': 0, 'tab': 0}
            
            for py_file in py_files:
                with open(py_file, 'r', encoding='utf-8', errors='ignore') as f:
                    content = f.read()
                    if '\t' in content[:1000]:
                        indent_counts['tab'] += 1
                    else:
                        indent_counts['space'] += 1
            
            style['indentation'] = max(indent_counts, key=indent_counts.get)
        
        return style
    
    def _find_key_files(self) -> List[Dict]:
        """查找关键文件"""
        key_patterns = [
            ('main', ['main.py', 'main.js', 'index.js', 'app.py']),
            ('config', ['config.py', 'settings.py', '.env', 'config.json']),
            ('model', ['models.py', 'schema.py']),
            ('api', ['api.py', 'routes.py', 'views.py']),
            ('test', ['test_*.py', '*_test.py', 'tests/*.py'])
        ]
        
        key_files = []
        for category, patterns in key_patterns:
            for pattern in patterns:
                matches = list(self.project_path.glob(f'**/{pattern}'))
                for match in matches[:3]:
                    key_files.append({
                        'category': category,
                        'path': str(match.relative_to(self.project_path)),
                        'name': match.name
                    })
        
        return key_files
    
    def _get_git_info(self) -> Dict:
        """获取 Git 信息"""
        try:
            import git
            repo = git.Repo(self.project_path)
            return {
                'branch': repo.active_branch.name,
                'last_commit': repo.head.commit.message.strip(),
                'last_commit_date': repo.head.commit.committed_datetime.isoformat(),
                'remote': repo.remotes.origin.url if repo.remotes else None
            }
        except:
            return {}
```

```python
# server/context/code_indexer.py

from typing import Dict, List, Any
from pathlib import Path
import hashlib
from datetime import datetime

class CodeIndexer:
    """代码索引器 - 构建代码知识库"""
    
    def __init__(self, embedding_model):
        self.embedding_model = embedding_model
        self.index = {}
    
    def build_index(self, project_info: Dict) -> Dict:
        """构建项目索引"""
        index = {
            'project_id': self._generate_project_id(project_info),
            'updated_at': datetime.now().isoformat(),
            'tech_stack': project_info['tech_stack'],
            'files': {},
            'symbols': {},
            'relationships': []
        }
        
        # 索引关键文件
        for key_file in project_info.get('key_files', []):
            file_index = self._index_file(key_file['path'])
            if file_index:
                index['files'][key_file['path']] = file_index
        
        # 提取项目概念
        index['concepts'] = self._extract_concepts(project_info)
        
        # 建立关系
        index['relationships'] = self._build_relationships(index)
        
        return index
    
    def _index_file(self, file_path: str) -> Dict:
        """索引单个文件"""
        try:
            full_path = Path(file_path)
            if not full_path.exists():
                return None
            
            with open(full_path, 'r', encoding='utf-8', errors='ignore') as f:
                content = f.read()
            
            # 生成嵌入向量（限制长度）
            embedding = self.embedding_model.embed(content[:5000])
            
            # 提取符号（类、函数等）
            symbols = self._extract_symbols(content, file_path)
            
            return {
                'path': file_path,
                'size': len(content),
                'lines': content.count('\n'),
                'embedding': embedding,
                'summary': self._summarize_file(content),
                'symbols': symbols,
                'updated_at': datetime.fromtimestamp(
                    full_path.stat().st_mtime
                ).isoformat()
            }
        except Exception as e:
            print(f"索引文件失败 {file_path}: {e}")
            return None
    
    def _extract_symbols(self, content: str, file_path: str) -> List[Dict]:
        """提取代码符号"""
        symbols = []
        ext = Path(file_path).suffix
        
        if ext == '.py':
            import re
            # 提取类
            for match in re.finditer(r'class\s+(\w+)', content):
                symbols.append({
                    'type': 'class',
                    'name': match.group(1),
                    'line': content[:match.start()].count('\n')
                })
            # 提取函数
            for match in re.finditer(r'def\s+(\w+)', content):
                symbols.append({
                    'type': 'function',
                    'name': match.group(1),
                    'line': content[:match.start()].count('\n')
                })
        
        elif ext in ['.js', '.ts', '.tsx']:
            import re
            # 提取函数
            for match in re.finditer(r'function\s+(\w+)', content):
                symbols.append({
                    'type': 'function',
                    'name': match.group(1),
                    'line': content[:match.start()].count('\n')
                })
            # 提取组件（React）
            for match in re.finditer(r'const\s+(\w+)\s*=\s*\([^)]*\)\s*=>', content):
                symbols.append({
                    'type': 'component',
                    'name': match.group(1),
                    'line': content[:match.start()].count('\n')
                })
        
        return symbols
    
    def _summarize_file(self, content: str) -> str:
        """生成文件摘要"""
        lines = content.split('\n')
        summary_lines = []
        
        # 提取文件头注释
        for line in lines[:20]:
            if line.strip().startswith('#') or line.strip().startswith('//'):
                summary_lines.append(line.strip())
        
        # 提取关键定义
        for line in lines[:100]:
            if any(kw in line for kw in ['class ', 'def ', 'function ', 'const ', 'export ']):
                summary_lines.append(line.strip())
        
        return '\n'.join(summary_lines[:10])
    
    def _extract_concepts(self, project_info: Dict) -> Dict:
        """提取项目概念"""
        return {
            'architecture': self._detect_architecture(project_info),
            'patterns': self._detect_patterns(project_info),
            'domain': self._detect_domain(project_info)
        }
    
    def _detect_architecture(self, project_info: Dict) -> str:
        """检测架构模式"""
        structure = project_info.get('structure', {})
        
        # MVC 模式
        if any('controllers' in str(c) for c in structure.get('children', [])):
            return 'MVC'
        
        # 分层架构
        if any('services' in str(c) for c in structure.get('children', [])):
            return 'Layered'
        
        # 微服务
        if len([c for c in structure.get('children', []) if c.get('type') == 'folder']) > 10:
            return 'Microservices'
        
        return 'Monolithic'
    
    def _detect_patterns(self, project_info: Dict) -> List[str]:
        """检测设计模式"""
        patterns = []
        
        # 简单检测
        tech_stack = project_info.get('tech_stack', [])
        if 'React' in tech_stack:
            patterns.extend(['Component', 'Props', 'State'])
        if 'FastAPI' in tech_stack:
            patterns.extend(['Dependency Injection', 'Router'])
        
        return patterns
    
    def _detect_domain(self, project_info: Dict) -> str:
        """检测项目领域"""
        # 根据文件名和依赖推测
        key_files = project_info.get('key_files', [])
        deps = project_info.get('dependencies', {})
        
        domain_keywords = {
            '电商': ['shop', 'cart', 'product', 'order'],
            '社交': ['user', 'friend', 'message', 'post'],
            '金融': ['payment', 'transaction', 'account'],
            '教育': ['course', 'student', 'lesson'],
            'AI/ML': ['model', 'train', 'predict', 'torch', 'tensorflow']
        }
        
        # 简单匹配
        all_text = ' '.join([f['name'].lower() for f in key_files])
        all_text += ' ' + ' '.join(deps.get('python', []))
        
        for domain, keywords in domain_keywords.items():
            if any(kw in all_text for kw in keywords):
                return domain
        
        return '通用'
    
    def _build_relationships(self, index: Dict) -> List[Dict]:
        """构建概念关系"""
        relationships = []
        
        # 文件之间的关系
        for path, file_data in index['files'].items():
            for symbol in file_data.get('symbols', []):
                relationships.append({
                    'type': 'defines',
                    'source': path,
                    'target': symbol['name'],
                    'symbol_type': symbol['type']
                })
        
        return relationships
    
    def _generate_project_id(self, project_info: Dict) -> str:
        """生成项目 ID"""
        content = f"{project_info['name']}:{project_info['path']}"
        return hashlib.md5(content.encode()).hexdigest()
```

### 第 3 步：创建上下文检索器

```python
# server/context/context_retriever.py

from typing import List, Dict, Any, Optional
import numpy as np

class ContextRetriever:
    """上下文检索器 - 智能检索相关代码"""
    
    def __init__(self, index: Dict, embedding_model):
        self.index = index
        self.embedding_model = embedding_model
    
    def retrieve(self, query: str, top_k: int = 5) -> List[Dict]:
        """检索与查询最相关的上下文"""
        # 生成查询向量
        query_embedding = self.embedding_model.embed(query)
        
        # 搜索相关文件
        file_scores = []
        for path, file_data in self.index['files'].items():
            if 'embedding' in file_data:
                similarity = self._cosine_similarity(
                    query_embedding,
                    file_data['embedding']
                )
                file_scores.append({
                    'path': path,
                    'score': similarity,
                    'data': file_data
                })
        
        # 按相似度排序
        file_scores.sort(key=lambda x: x['score'], reverse=True)
        
        # 返回 top_k
        results = []
        for item in file_scores[:top_k]:
            results.append({
                'type': 'file',
                'path': item['path'],
                'relevance': item['score'],
                'summary': item['data'].get('summary', ''),
                'symbols': item['data'].get('symbols', [])
            })
        
        # 添加项目级信息
        results.append({
            'type': 'project',
            'tech_stack': self.index['tech_stack'],
            'architecture': self.index['concepts'].get('architecture', ''),
            'domain': self.index['concepts'].get('domain', '')
        })
        
        return results
    
    def retrieve_for_code_completion(
        self,
        file_path: str,
        cursor_position: int,
        content: str
    ) -> Dict:
        """为代码补全检索上下文"""
        # 获取当前文件
        current_file = self.index['files'].get(file_path, {})
        
        # 获取相关函数/类
        related_symbols = self._find_related_symbols(content, cursor_position)
        
        # 获取导入的模块
        imports = self._parse_imports(content)
        
        return {
            'current_file': current_file.get('summary', ''),
            'related_symbols': related_symbols,
            'imports': imports,
            'project_context': {
                'tech_stack': self.index['tech_stack'],
                'architecture': self.index['concepts'].get('architecture', '')
            }
        }
    
    def _cosine_similarity(self, v1: List[float], v2: List[float]) -> float:
        """计算余弦相似度"""
        v1_arr = np.array(v1)
        v2_arr = np.array(v2)
        return np.dot(v1_arr, v2_arr) / (np.linalg.norm(v1_arr) * np.linalg.norm(v2_arr))
    
    def _find_related_symbols(self, content: str, position: int) -> List[Dict]:
        """查找相关符号"""
        import re
        words = re.findall(r'\b(\w+)\b', content[:position])
        if words:
            last_word = words[-1]
            # 在索引中查找相关符号
            related = []
            for path, file_data in self.index['files'].items():
                for symbol in file_data.get('symbols', []):
                    if symbol['name'] == last_word:
                        related.append({
                            'name': symbol['name'],
                            'type': symbol['type'],
                            'file': path
                        })
            return related
        return []
    
    def _parse_imports(self, content: str) -> List[str]:
        """解析导入语句"""
        import re
        imports = []
        
        # Python imports
        for match in re.finditer(r'^import\s+([\w.]+)', content, re.MULTILINE):
            imports.append(match.group(1))
        for match in re.finditer(r'^from\s+([\w.]+)\s+import', content, re.MULTILINE):
            imports.append(match.group(1))
        
        return imports
```

```python
# server/api/context.py

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel, Field
from typing import Dict, Any, Optional, List

from context.project_scanner import ProjectScanner
from context.code_indexer import CodeIndexer
from context.context_retriever import ContextRetriever
from rag.embedder import get_embedder

router = APIRouter(prefix="/context", tags=["项目上下文"])

# 全局缓存
project_cache: Dict[str, Dict] = {}
index_cache: Dict[str, Dict] = {}

class ScanRequest(BaseModel):
    project_path: str

class RetrieveRequest(BaseModel):
    query: str
    project_path: Optional[str] = None
    top_k: int = 5

class CodeCompletionRequest(BaseModel):
    file_path: str
    content: str
    cursor_position: int

@router.post("/scan")
async def scan_project(request: ScanRequest):
    """扫描项目"""
    try:
        scanner = ProjectScanner(request.project_path)
        project_info = scanner.scan()
        
        # 缓存
        project_cache[request.project_path] = project_info
        
        # 构建索引
        embedder = get_embedder()
        indexer = CodeIndexer(embedder)
        index = indexer.build_index(project_info)
        index_cache[request.project_path] = index
        
        return {
            "success": True,
            "project": project_info,
            "index_summary": {
                "files_indexed": len(index['files']),
                "symbols_found": sum(
                    len(f.get('symbols', []))
                    for f in index['files'].values()
                )
            }
        }
    except Exception as e:
        raise HTTPException(500, f"扫描失败：{str(e)}")

@router.post("/retrieve")
async def retrieve_context(request: RetrieveRequest):
    """检索项目上下文"""
    try:
        project_path = request.project_path or list(index_cache.keys())[0]
        
        if project_path not in index_cache:
            # 自动扫描
            await scan_project(ScanRequest(project_path=project_path))
        
        index = index_cache[project_path]
        embedder = get_embedder()
        retriever = ContextRetriever(index, embedder)
        
        results = retriever.retrieve(request.query, request.top_k)
        
        return {
            "success": True,
            "context": results,
            "project_info": project_cache.get(project_path, {})
        }
    except Exception as e:
        raise HTTPException(500, f"检索失败：{str(e)}")

@router.post("/code-completion")
async def code_completion_context(request: CodeCompletionRequest):
    """获取代码补全上下文"""
    try:
        # 找到项目路径
        project_path = None
        for path in index_cache.keys():
            if request.file_path.startswith(path):
                project_path = path
                break
        
        if not project_path:
            return {
                "success": False,
                "error": "文件不在已索引的项目中"
            }
        
        index = index_cache[project_path]
        embedder = get_embedder()
        retriever = ContextRetriever(index, embedder)
        
        context = retriever.retrieve_for_code_completion(
            request.file_path,
            request.cursor_position,
            request.content
        )
        
        return {
            "success": True,
            "context": context
        }
    except Exception as e:
        raise HTTPException(500, f"获取上下文失败：{str(e)}")

@router.get("/projects")
async def list_projects():
    """列出已索引的项目"""
    return {
        "success": True,
        "projects": [
            {
                "path": path,
                "name": info.get('name', ''),
                "tech_stack": info.get('tech_stack', []),
                "indexed_at": index_cache[path].get('updated_at', '')
            }
            for path, info in project_cache.items()
        ]
    }

@router.delete("/project/{project_path:path}")
async def remove_project(project_path: str):
    """移除项目索引"""
    if project_path in project_cache:
        del project_cache[project_path]
    if project_path in index_cache:
        del index_cache[project_path]
    
    return {"success": True, "message": "已移除"}
```

```python
# server/api/chat.py (修改)

from context.context_retriever import ContextRetriever
from context.code_indexer import CodeIndexer
from context.project_scanner import ProjectScanner

@router.post("/chat")
async def chat(request: ChatRequest):
    """聊天接口（带项目上下文）"""
    
    # 1. 检索项目上下文
    context = ""
    if request.use_context:
        try:
            # 获取项目路径（从配置或默认）
            project_path = get_default_project_path()
            
            if project_path not in index_cache:
                # 自动扫描
                scanner = ProjectScanner(project_path)
                project_info = scanner.scan()
                embedder = get_embedder()
                indexer = CodeIndexer(embedder)
                index = indexer.build_index(project_info)
                index_cache[project_path] = index
                project_cache[project_path] = project_info
            
            # 检索
            index = index_cache[project_path]
            embedder = get_embedder()
            retriever = ContextRetriever(index, embedder)
            
            results = retriever.retrieve(request.messages[-1]['content'], top_k=3)
            
            # 组装上下文
            context_parts = []
            for result in results:
                if result['type'] == 'file':
                    context_parts.append(
                        f"相关文件：{result['path']}\n"
                        f"摘要：{result['summary']}\n"
                    )
                elif result['type'] == 'project':
                    context_parts.append(
                        f"项目技术栈：{', '.join(result['tech_stack'])}\n"
                        f"架构：{result['architecture']}\n"
                    )
            
            context = "\n".join(context_parts)
        except Exception as e:
            print(f"获取项目上下文失败：{e}")
    
    # 2. 构建带上下文的 prompt
    if context:
        system_prompt = f"""你是一个有帮助的 AI 助手。

项目上下文：
{context}

请根据以上项目信息，给用户一个有帮助的回答。
如果问题与项目相关，请考虑项目的技术栈、架构和代码风格。
"""
    else:
        system_prompt = "你是一个有帮助的 AI 助手。"
    
    # 3. 调用模型
    model = get_model(request.model)
    response = model.generate(
        messages=request.messages,
        system_prompt=system_prompt
    )
    
    return {"content": response, "context_used": bool(context)}
```

```tsx
// client/src/pages/ProjectContext.tsx

import { useState, useEffect } from 'react'
import { Card, Button, List, Tag, Space, Input, Spin, Alert } from 'antd'
import { FolderOutlined, CodeOutlined, SearchOutlined } from '@ant-design/icons'

export const ProjectContext: React.FC = () => {
  const [projects, setProjects] = useState([])
  const [loading, setLoading] = useState(false)
  const [searchPath, setSearchPath] = useState('')
  const [context, setContext] = useState(null)

  const scanProject = async () => {
    setLoading(true)
    try {
      const response = await fetch('http://127.0.0.1:8000/context/scan', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ project_path: searchPath })
      })
      const data = await response.json()
      if (data.success) {
        setContext(data.project)
        loadProjects()
      }
    } catch (error) {
      console.error('扫描失败:', error)
    } finally {
      setLoading(false)
    }
  }

  const loadProjects = async () => {
    const response = await fetch('http://127.0.0.1:8000/context/projects')
    const data = await response.json()
    setProjects(data.projects)
  }

  useEffect(() => {
    loadProjects()
  }, [])

  return (
    <div style={{ padding: 24 }}>
      <Card title="📁 项目上下文管理" style={{ marginBottom: 16 }}>
        <Space.Compact style={{ width: '100%' }}>
          <Input
            placeholder="项目路径，例如：C:/Users/JHJ/Desktop/finetune-platform"
            value={searchPath}
            onChange={(e) => setSearchPath(e.target.value)}
          />
          <Button
            type="primary"
            icon={<FolderOutlined />}
            onClick={scanProject}
            loading={loading}
          >
            扫描项目
          </Button>
        </Space.Compact>
      </Card>

      {context && (
        <Card title="项目信息" style={{ marginBottom: 16 }}>
          <List
            dataSource={[
              { label: '项目名称', value: context.name },
              { label: '技术栈', value: context.tech_stack?.join(', ') },
              { label: '架构', value: context.architecture },
              { label: '领域', value: context.domain },
              { label: 'Git 分支', value: context.git_info?.branch },
            ]}
            renderItem={(item) => (
              <List.Item>
                <List.Item.Meta
                  title={item.label}
                  description={item.value || '未知'}
                />
              </List.Item>
            )}
          />
        </Card>
      )}

      {projects.length > 0 && (
        <Card title="已索引的项目">
          <List
            dataSource={projects}
            renderItem={(project) => (
              <List.Item
                actions={[
                  <Button
                    size="small"
                    danger
                    onClick={() => removeProject(project.path)}
                  >
                    移除
                  </Button>
                ]}
              >
                <List.Item.Meta
                  title={
                    <Space>
                      <CodeOutlined />
                      {project.name}
                    </Space>
                  }
                  description={
                    <Space wrap>
                      {project.tech_stack?.map(tech => (
                        <Tag key={tech}>{tech}</Tag>
                      ))}
                    </Space>
                  }
                />
              </List.Item>
            )}
          />
        </Card>
      )}
    </div>
  )
}
```
