### 第 1 步：创建文件操作 API

```python
# server/api/file_operations.py

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any
from pathlib import Path
import os

router = APIRouter(prefix="/file", tags=["文件操作"])

# 工作目录（项目根目录）
WORKING_DIR = Path(__file__).parent.parent.parent


class CreateFileRequest(BaseModel):
    file_path: str
    content: str = ""
    overwrite: bool = False


class WriteFileRequest(BaseModel):
    file_path: str
    content: str
    mode: str = "w"  # w: 覆盖，a: 追加


class ReadFileRequest(BaseModel):
    file_path: str
    max_lines: Optional[int] = None


class DeleteFileRequest(BaseModel):
    file_path: str


class ListFilesRequest(BaseModel):
    directory: str = "."
    pattern: str = "*"
    include_subdirs: bool = False


class FileResponse(BaseModel):
    success: bool
    message: str = ""
    data: Optional[Any] = None
    error: Optional[str] = None


def safe_path(file_path: str) -> Path:
    """安全检查文件路径，防止目录遍历攻击"""
    # 如果是绝对路径
    if os.path.isabs(file_path):
        path = Path(file_path)
    else:
        # 相对路径，限制在工作目录内
        path = (WORKING_DIR / file_path).resolve()
    
    # 确保路径在工作目录内
    try:
        path.relative_to(WORKING_DIR)
    except ValueError:
        raise HTTPException(403, f"禁止访问工作目录外的文件：{file_path}")
    
    return path


@router.post("/create", response_model=FileResponse)
async def create_file(request: CreateFileRequest):
    """创建文件"""
    try:
        file_path = safe_path(request.file_path)
        
        # 检查文件是否已存在
        if file_path.exists() and not request.overwrite:
            return FileResponse(
                success=False,
                error=f"文件已存在：{request.file_path}",
                data={"exists": True}
            )
        
        # 创建父目录
        file_path.parent.mkdir(parents=True, exist_ok=True)
        
        # 创建文件
        with open(file_path, 'w', encoding='utf-8') as f:
            f.write(request.content)
        
        return FileResponse(
            success=True,
            message=f"文件已创建：{request.file_path}",
            data={
                "path": str(file_path),
                "size": len(request.content),
                "absolute_path": str(file_path.absolute())
            }
        )
    except HTTPException:
        raise
    except Exception as e:
        return FileResponse(
            success=False,
            error=f"创建失败：{str(e)}"
        )


@router.post("/write", response_model=FileResponse)
async def write_file(request: WriteFileRequest):
    """写入文件"""
    try:
        file_path = safe_path(request.file_path)
        
        if not file_path.exists():
            return FileResponse(
                success=False,
                error=f"文件不存在：{request.file_path}"
            )
        
        # 写入内容
        with open(file_path, request.mode, encoding='utf-8') as f:
            f.write(request.content)
        
        return FileResponse(
            success=True,
            message=f"文件已{'追加' if request.mode == 'a' else '更新'}：{request.file_path}",
            data={
                "path": str(file_path),
                "size": file_path.stat().st_size
            }
        )
    except Exception as e:
        return FileResponse(
            success=False,
            error=f"写入失败：{str(e)}"
        )


@router.post("/read", response_model=FileResponse)
async def read_file(request: ReadFileRequest):
    """读取文件"""
    try:
        file_path = safe_path(request.file_path)
        
        if not file_path.exists():
            return FileResponse(
                success=False,
                error=f"文件不存在：{request.file_path}"
            )
        
        if not file_path.is_file():
            return FileResponse(
                success=False,
                error=f"不是文件：{request.file_path}"
            )
        
        # 读取内容
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # 限制行数
        if request.max_lines:
            lines = content.split('\n')[:request.max_lines]
            content = '\n'.join(lines)
        
        return FileResponse(
            success=True,
            message=f"文件已读取：{request.file_path}",
            data={
                "path": str(file_path),
                "content": content,
                "size": file_path.stat().st_size,
                "lines": len(content.split('\n'))
            }
        )
    except Exception as e:
        return FileResponse(
            success=False,
            error=f"读取失败：{str(e)}"
        )


@router.post("/delete", response_model=FileResponse)
async def delete_file(request: DeleteFileRequest):
    """删除文件"""
    try:
        file_path = safe_path(request.file_path)
        
        if not file_path.exists():
            return FileResponse(
                success=False,
                error=f"文件不存在：{request.file_path}"
            )
        
        if not file_path.is_file():
            return FileResponse(
                success=False,
                error=f"不能删除目录：{request.file_path}"
            )
        
        # 删除文件
        file_path.unlink()
        
        return FileResponse(
            success=True,
            message=f"文件已删除：{request.file_path}"
        )
    except Exception as e:
        return FileResponse(
            success=False,
            error=f"删除失败：{str(e)}"
        )


@router.post("/list", response_model=FileResponse)
async def list_files(request: ListFilesRequest):
    """列出文件"""
    try:
        dir_path = safe_path(request.directory)
        
        if not dir_path.exists():
            return FileResponse(
                success=False,
                error=f"目录不存在：{request.directory}"
            )
        
        if not dir_path.is_dir():
            return FileResponse(
                success=False,
                error=f"不是目录：{request.directory}"
            )
        
        # 列出文件
        files = []
        if request.include_subdirs:
            for item in dir_path.rglob(request.pattern):
                if item.is_file():
                    files.append({
                        "name": item.name,
                        "path": str(item.relative_to(WORKING_DIR)),
                        "absolute_path": str(item.absolute()),
                        "is_dir": item.is_dir(),
                        "size": item.stat().st_size if item.is_file() else 0
                    })
        else:
            for item in dir_path.glob(request.pattern):
                files.append({
                    "name": item.name,
                    "path": str(item.relative_to(WORKING_DIR)),
                    "absolute_path": str(item.absolute()),
                    "is_dir": item.is_dir(),
                    "size": item.stat().st_size if item.is_file() else 0
                })
        
        return FileResponse(
            success=True,
            message=f"找到 {len(files)} 个文件",
            data={
                "directory": str(dir_path),
                "count": len(files),
                "files": files
            }
        )
    except Exception as e:
        return FileResponse(
            success=False,
            error=f"列出失败：{str(e)}"
        )
```

@router.post("/detect-intent")
async def detect_intent(request: IntentRequest):
    """检测消息是否需要执行 Agent 操作"""
    
    message = request.message.lower()
    
    # ⭐ 文件操作模式
    file_patterns = [
        # 创建文件
        (r'创建\s+(\S+)\s*文件', 'file_create', lambda m: {
            'file_path': m.group(1),
            'content': ''
        }),
        (r'新建\s+(\S+)', 'file_create', lambda m: {
            'file_path': m.group(1),
            'content': ''
        }),
        
        # 写入文件
        (r'把\s+(\S+)\s+的内容改成\s+(.+)', 'file_write', lambda m: {
            'file_path': m.group(1),
            'content': m.group(2)
        }),
        (r'写入\s+(\S+)\s*(.*)', 'file_write', lambda m: {
            'file_path': m.group(1),
            'content': m.group(2) or ''
        }),
        
        # 读取文件
        (r'读取\s+(\S+)', 'file_read', lambda m: {
            'file_path': m.group(1)
        }),
        (r'查看\s+(\S+)\s*的内容', 'file_read', lambda m: {
            'file_path': m.group(1)
        }),
        
        # 删除文件
        (r'删除\s+(\S+)', 'file_delete', lambda m: {
            'file_path': m.group(1)
        }),
        
        # 列出文件
        (r'列出\s*(.+?)\s*的文件', 'file_list', lambda m: {
            'directory': m.group(1) or '.'
        }),
    ]

    for pattern, action, params_extractor in file_patterns:

# server/agent/executor.py (修改)

class ComputerOperator:
    """电脑操作器"""
    
    def __init__(self):
        self.working_dir = Path(__file__).parent.parent.parent
        
        self.actions = {
            # ⭐ 新增文件操作
            'file_create': self.file_create,
            'file_write': self.file_write,
            'file_read': self.file_read,
            'file_delete': self.file_delete,
            'file_list': self.file_list,
        }
    
    async def file_create(self, file_path: str, content: str = "", overwrite: bool = False) -> Dict:
        """创建文件"""
        path = self._safe_path(file_path)
        
        if path.exists() and not overwrite:
            return {"success": False, "error": f"文件已存在：{file_path}"}
        
        path.parent.mkdir(parents=True, exist_ok=True)
        
        with open(path, 'w', encoding='utf-8') as f:
            f.write(content)
        
        return {
            "success": True,
            "message": f"已创建：{file_path}",
            "path": str(path.absolute())
        }
    
    async def file_write(self, file_path: str, content: str, mode: str = "w") -> Dict:
        """写入文件"""
        path = self._safe_path(file_path)
        
        if not path.exists():
            return {"success": False, "error": f"文件不存在：{file_path}"}
        
        with open(path, mode, encoding='utf-8') as f:
            f.write(content)
        
        return {"success": True, "message": f"已{'追加' if mode == 'a' else '更新'}：{file_path}"}
    
    async def file_read(self, file_path: str, max_lines: int = None) -> Dict:
        """读取文件"""
        path = self._safe_path(file_path)
        
        if not path.exists():
            return {"success": False, "error": f"文件不存在：{file_path}"}
        
        with open(path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        if max_lines:
            lines = content.split('\n')[:max_lines]
            content = '\n'.join(lines)
        
        return {"success": True, "content": content, "path": str(path.absolute())}
    
    async def file_delete(self, file_path: str) -> Dict:
        """删除文件"""
        path = self._safe_path(file_path)
        
        if not path.exists():
            return {"success": False, "error": f"文件不存在：{file_path}"}
        
        if not path.is_file():
            return {"success": False, "error": "不能删除目录"}
        
        path.unlink()
        
        return {"success": True, "message": f"已删除：{file_path}"}
    
    async def file_list(self, directory: str = ".", pattern: str = "*") -> Dict:
        """列出文件"""
        dir_path = self._safe_path(directory)
        
        if not dir_path.exists():
            return {"success": False, "error": f"目录不存在：{directory}"}
        
        files = []
        for item in dir_path.glob(pattern):
            files.append({
                "name": item.name,
                "path": str(item.relative_to(self.working_dir)),
                "is_dir": item.is_dir()
            })
        
        return {"success": True, "count": len(files), "files": files}
    
    def _safe_path(self, file_path: str) -> Path:
        """安全检查文件路径"""
        if os.path.isabs(file_path):
            path = Path(file_path)
        else:
            path = (self.working_dir / file_path).resolve()
        
        try:
            path.relative_to(self.working_dir)
        except ValueError:
            raise Exception(f"禁止访问工作目录外的文件：{file_path}")
        
        return path

// client/src/pages/Chat.tsx (修改)

const executeAgent = async (action: string, params: any) => {
  setAgentExecution({
    status: 'executing',
    action,
    params,
    progress: 0
  })

  try {
    // ⭐ 文件操作特殊处理
    if (action === 'file_operation') {
      const operation = params.operation
      const endpoint = `http://127.0.0.1:8000/file/${operation}`
      
      const response = await fetch(endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(params)
      })

      const result = await response.json()

      if (result.success) {
        setAgentExecution({
          status: 'completed',
          action,
          params,
          progress: 100
        })

        setTimeout(() => setAgentExecution(null), 2000)

        return {
          success: true,
          result: result.data
        }
      } else {
        setAgentExecution({
          status: 'failed',
          action,
          params,
          progress: 0,
          error: result.error
        })

        return {
          success: false,
          error: result.error
        }
      }
    }
    
    // 其他操作...
  } catch (error) {
    return { success: false, error: String(error) }
  }
}
