```python
# server/context/project_scanner.py

from pathlib import Path
from typing import List, Dict
import git
import hashlib

class ProjectScanner:
    """项目扫描器 - 智能识别项目结构"""
    
    # 支持的技术栈
    TECH_STACKS = {
        'python': {
            'files': ['requirements.txt', 'setup.py', 'pyproject.toml'],
            'extensions': ['.py'],
            'ignore': ['__pycache__', '*.pyc', '.git']
        },
        'javascript': {
            'files': ['package.json', 'tsconfig.json'],
            'extensions': ['.js', '.ts', '.jsx', '.tsx'],
            'ignore': ['node_modules', 'dist', 'build']
        },
        'java': {
            'files': ['pom.xml', 'build.gradle'],
            'extensions': ['.java'],
            'ignore': ['target', 'build', '.class']
        }
    }
    
    def scan(self, project_path: str) -> Dict:
        """扫描项目，返回项目信息"""
        path = Path(project_path)
        
        project_info = {
            'name': path.name,
            'path': str(path),
            'tech_stack': self._detect_tech_stack(path),
            'structure': self._build_structure(path),
            'dependencies': self._parse_dependencies(path),
            'git_info': self._get_git_info(path),
            'code_style': self._analyze_code_style(path),
            'key_files': self._find_key_files(path)
        }
        
        return project_info
    
    def _detect_tech_stack(self, path: Path) -> List[str]:
        """检测技术栈"""
        stacks = []
        for stack, config in self.TECH_STACKS.items():
            for file in config['files']:
                if (path / file).exists():
                    stacks.append(stack)
                    break
        return stacks
    
    def _build_structure(self, path: Path, depth: int = 3) -> Dict:
        """构建项目结构树"""
        structure = {'name': path.name, 'type': 'folder', 'children': []}
        
        if depth <= 0:
            return structure
        
        try:
            for item in sorted(path.iterdir()):
                # 跳过忽略的目录
                if any(ign in str(item) for ign in ['__pycache__', 'node_modules', '.git']):
                    continue
                
                if item.is_file():
                    structure['children'].append({
                        'name': item.name,
                        'type': 'file',
                        'size': item.stat().st_size,
                        'path': str(item)
                    })
                else:
                    child_structure = self._build_structure(item, depth - 1)
                    if child_structure['children']:  # 只添加有内容的目录
                        structure['children'].append(child_structure)
        except PermissionError:
            pass
        
        return structure
    
    def _parse_dependencies(self, path: Path) -> Dict:
        """解析项目依赖"""
        deps = {}
        
        # Python
        req_file = path / 'requirements.txt'
        if req_file.exists():
            deps['python'] = self._parse_requirements(req_file)
        
        # JavaScript
        pkg_file = path / 'package.json'
        if pkg_file.exists():
            import json
            with open(pkg_file) as f:
                pkg_data = json.load(f)
                deps['javascript'] = {
                    'dependencies': pkg_data.get('dependencies', {}),
                    'devDependencies': pkg_data.get('devDependencies', {})
                }
        
        return deps
    
    def _analyze_code_style(self, path: Path) -> Dict:
        """分析代码风格"""
        style = {
            'indentation': 'space',  # or 'tab'
            'quote': 'single',  # or 'double'
            'naming': 'camelCase',  # or 'snake_case', 'PascalCase'
            'line_length': 88
        }
        
        # 分析 Python 文件
        py_files = list(path.glob('**/*.py'))[:10]  # 分析前 10 个文件
        if py_files:
            import re
            indent_counts = {'space': 0, 'tab': 0}
            
            for py_file in py_files:
                with open(py_file, 'r', encoding='utf-8', errors='ignore') as f:
                    content = f.read()
                    # 检测缩进
                    if '\t' in content[:1000]:
                        indent_counts['tab'] += 1
                    else:
                        indent_counts['space'] += 1
            
            style['indentation'] = max(indent_counts, key=indent_counts.get)
        
        return style
    
    def _find_key_files(self, path: Path) -> List[Dict]:
        """查找关键文件"""
        key_patterns = [
            ('main', ['main.py', 'main.js', 'index.js', 'app.py']),
            ('config', ['config.py', 'settings.py', '.env', 'config.json']),
            ('model', ['models.py', 'schema.py', 'models/*.py']),
            ('api', ['api.py', 'routes.py', 'views.py', 'controllers/*.py']),
            ('test', ['test_*.py', '*_test.py', 'tests/*.py'])
        ]
        
        key_files = []
        for category, patterns in key_patterns:
            for pattern in patterns:
                matches = list(path.glob(f'**/{pattern}'))
                for match in matches[:3]:  # 每类最多 3 个
                    key_files.append({
                        'category': category,
                        'path': str(match),
                        'name': match.name
                    })
        
        return key_files
    
    def _get_git_info(self, path: Path) -> Dict:
        """获取 Git 信息"""
        try:
            repo = git.Repo(path)
            return {
                'branch': repo.active_branch.name,
                'last_commit': repo.head.commit.message.strip(),
                'last_commit_date': repo.head.commit.committed_datetime,
                'remote': repo.remotes.origin.url if repo.remotes else None
            }
        except:
            return {}
```

##### 2. 索引构建器

```python
# server/context/indexer.py

from typing import List, Dict
import hashlib
from datetime import datetime

class ContextIndexer:
    """上下文索引器 - 构建项目知识索引"""
    
    def __init__(self, embedding_model):
        self.embedding_model = embedding_model
        self.index = {}  # 文件路径 → 索引数据
    
    def build_index(self, project_info: Dict) -> Dict:
        """构建项目索引"""
        index = {
            'project_id': self._generate_project_id(project_info),
            'updated_at': datetime.now().isoformat(),
            'tech_stack': project_info['tech_stack'],
            'structure_hash': self._hash_structure(project_info['structure']),
            'files': {},
            'concepts': {},
            'relationships': []
        }
        
        # 索引关键文件
        for key_file in project_info['key_files']:
            file_index = self._index_file(key_file['path'])
            if file_index:
                index['files'][key_file['path']] = file_index
        
        # 提取项目概念
        index['concepts'] = self._extract_concepts(project_info)
        
        # 建立关系
        index['relationships'] = self._build_relationships(index)
        
        return index
    
    def _index_file(self, file_path: str) -> Dict:
        """索引单个文件"""
        try:
            with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                content = f.read()
            
            # 生成嵌入向量
            embedding = self.embedding_model.embed(content[:5000])  # 限制长度
            
            # 提取元数据
            lines = content.split('\n')
            
            return {
                'path': file_path,
                'size': len(content),
                'lines': len(lines),
                'embedding': embedding,
                'summary': self._summarize_file(content),
                'symbols': self._extract_symbols(content, file_path),
                'updated_at': datetime.fromtimestamp(
                    Path(file_path).stat().st_mtime
                ).isoformat()
            }
        except Exception as e:
            print(f"索引文件失败 {file_path}: {e}")
            return None
    
    def _extract_concepts(self, project_info: Dict) -> Dict:
        """提取项目概念"""
        concepts = {
            'architecture': self._detect_architecture(project_info),
            'patterns': self._detect_patterns(project_info),
            'domain': self._detect_domain(project_info)
        }
        return concepts
    
    def _detect_architecture(self, project_info: Dict) -> str:
        """检测架构模式"""
        structure = project_info['structure']
        
        # MVC 模式
        if any('controllers' in str(c) for c in structure.get('children', [])):
            return 'MVC'
        
        # 分层架构
        if any('services' in str(c) for c in structure.get('children', [])):
            return 'Layered'
        
        # 微服务
        if len([c for c in structure.get('children', []) if c.get('type') == 'folder']) > 10:
            return 'Microservices'
        
        return 'Monolithic'
    
    def _summarize_file(self, content: str) -> str:
        """生成文件摘要"""
        # 简单实现：提取前几行注释和关键定义
        lines = content.split('\n')
        summary_lines = []
        
        # 提取文件头注释
        for line in lines[:20]:
            if line.strip().startswith('#') or line.strip().startswith('//'):
                summary_lines.append(line.strip())
        
        # 提取关键定义
        for line in lines[:100]:
            if any(kw in line for kw in ['class ', 'def ', 'function ', 'const ']):
                summary_lines.append(line.strip())
        
        return '\n'.join(summary_lines[:10])
    
    def _extract_symbols(self, content: str, file_path: str) -> List[Dict]:
        """提取代码符号"""
        symbols = []
        ext = Path(file_path).suffix
        
        if ext == '.py':
            import re
            # 提取类和函数
            for match in re.finditer(r'class\s+(\w+)', content):
                symbols.append({
                    'type': 'class',
                    'name': match.group(1),
                    'line': content[:match.start()].count('\n')
                })
            for match in re.finditer(r'def\s+(\w+)', content):
                symbols.append({
                    'type': 'function',
                    'name': match.group(1),
                    'line': content[:match.start()].count('\n')
                })
        
        return symbols
    
    def _generate_project_id(self, project_info: Dict) -> str:
        """生成项目 ID"""
        content = f"{project_info['name']}:{project_info['path']}"
        return hashlib.md5(content.encode()).hexdigest()
    
    def _hash_structure(self, structure: Dict) -> str:
        """生成结构哈希"""
        import json
        return hashlib.md5(json.dumps(structure, sort_keys=True).encode()).hexdigest()
    
    def _build_relationships(self, index: Dict) -> List[Dict]:
        """构建概念关系"""
        relationships = []
        
        # 文件之间的关系
        for path, file_data in index['files'].items():
            for symbol in file_data.get('symbols', []):
                relationships.append({
                    'type': 'defines',
                    'source': path,
                    'target': symbol['name'],
                    'symbol_type': symbol['type']
                })
        
        return relationships
```

##### 3. 上下文检索器

```python
# server/context/retriever.py

from typing import List, Dict, Optional
import numpy as np

class ContextRetriever:
    """上下文检索器 - 智能检索相关代码和文档"""
    
    def __init__(self, index: Dict, embedding_model):
        self.index = index
        self.embedding_model = embedding_model
    
    def retrieve(self, query: str, top_k: int = 5) -> List[Dict]:
        """检索与查询最相关的上下文"""
        # 生成查询向量
        query_embedding = self.embedding_model.embed(query)
        
        # 搜索相关文件
        file_scores = []
        for path, file_data in self.index['files'].items():
            if 'embedding' in file_data:
                similarity = self._cosine_similarity(
                    query_embedding,
                    file_data['embedding']
                )
                file_scores.append({
                    'path': path,
                    'score': similarity,
                    'data': file_data
                })
        
        # 按相似度排序
        file_scores.sort(key=lambda x: x['score'], reverse=True)
        
        # 返回 top_k
        results = []
        for item in file_scores[:top_k]:
            results.append({
                'type': 'file',
                'path': item['path'],
                'relevance': item['score'],
                'summary': item['data'].get('summary', ''),
                'symbols': item['data'].get('symbols', [])
            })
        
        # 添加项目级信息
        results.append({
            'type': 'project',
            'tech_stack': self.index['tech_stack'],
            'architecture': self.index['concepts'].get('architecture', ''),
            'dependencies': self.index.get('dependencies', {})
        })
        
        return results
    
    def retrieve_for_code_completion(
        self,
        file_path: str,
        cursor_position: int,
        content: str
    ) -> Dict:
        """为代码补全检索上下文"""
        # 获取当前文件
        current_file = self.index['files'].get(file_path, {})
        
        # 获取相关函数/类
        related_symbols = self._find_related_symbols(content, cursor_position)
        
        # 获取导入的模块
        imports = self._parse_imports(content)
        
        return {
            'current_file': current_file.get('summary', ''),
            'related_symbols': related_symbols,
            'imports': imports,
            'project_context': {
                'tech_stack': self.index['tech_stack'],
                'architecture': self.index['concepts'].get('architecture', '')
            }
        }
    
    def _cosine_similarity(self, v1: List[float], v2: List[float]) -> float:
        """计算余弦相似度"""
        v1_arr = np.array(v1)
        v2_arr = np.array(v2)
        return np.dot(v1_arr, v2_arr) / (np.linalg.norm(v1_arr) * np.linalg.norm(v2_arr))
    
    def _find_related_symbols(self, content: str, position: int) -> List[Dict]:
        """查找相关符号"""
        # 简单实现：查找当前位置附近的函数/类名
        import re
        words = re.findall(r'\b(\w+)\b', content[:position])
        if words:
            last_word = words[-1]
            # 在索引中查找相关符号
            related = []
            for path, file_data in self.index['files'].items():
                for symbol in file_data.get('symbols', []):
                    if symbol['name'] == last_word:
                        related.append({
                            'name': symbol['name'],
                            'type': symbol['type'],
                            'file': path
                        })
            return related
        return []
    
    def _parse_imports(self, content: str) -> List[str]:
        """解析导入语句"""
        import re
        imports = []
        
        # Python imports
        for match in re.finditer(r'^import\s+([\w.]+)', content, re.MULTILINE):
            imports.append(match.group(1))
        for match in re.finditer(r'^from\s+([\w.]+)\s+import', content, re.MULTILINE):
            imports.append(match.group(1))
        
        return imports
```

```python
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import List, Optional, Dict

router = APIRouter(prefix="/api/context", tags=["context"])

class ProjectContextRequest(BaseModel):
    project_path: str
    query: Optional[str] = None
    top_k: int = 5

class CodeCompletionRequest(BaseModel):
    file_path: str
    content: str
    cursor_position: int

@router.post("/project")
async def get_project_context(request: ProjectContextRequest):
    """获取项目上下文"""
    try:
        # 扫描项目
        scanner = ProjectScanner()
        project_info = scanner.scan(request.project_path)
        
        # 构建索引
        indexer = ContextIndexer(embedding_model)
        index = indexer.build_index(project_info)
        
        # 检索相关上下文
        if request.query:
            retriever = ContextRetriever(index, embedding_model)
            context = retriever.retrieve(request.query, request.top_k)
        else:
            context = {'project_info': project_info, 'index': index}
        
        return {
            'success': True,
            'context': context
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/code-completion")
async def get_code_completion_context(request: CodeCompletionRequest):
    """获取代码补全上下文"""

        context = retriever.retrieve_for_code_completion(
            request.file_path,
            request.cursor_position,
            request.content
        )
        
        return {
            'success': True,
            'context': context
        }
```

```python
# server/workflow/engine.py

from typing import Dict, List, Any, Optional
from datetime import datetime
import asyncio
import json

class WorkflowEngine:
    """工作流执行引擎"""
    
    def __init__(self):
        self.nodes = {}  # 节点注册表
        self.workflows = {}  # 工作流实例
        self.register_builtin_nodes()
    
    def register_builtin_nodes(self):
        """注册内置任务节点"""
        # 文件操作节点
        self.register_node('read_file', self.node_read_file)
        self.register_node('write_file', self.node_write_file)
        self.register_node('move_file', self.node_move_file)
        self.register_node('delete_file', self.node_delete_file)
        self.register_node('list_files', self.node_list_files)
        
        # 数据处理节点
        self.register_node('parse_csv', self.node_parse_csv)
        self.register_node('parse_json', self.node_parse_json)
        self.register_node('transform_data', self.node_transform_data)
        
        # AI 节点
        self.register_node('ai_analyze', self.node_ai_analyze)
        self.register_node('ai_generate', self.node_ai_generate)
        
        # 控制节点
        self.register_node('condition', self.node_condition)
        self.register_node('loop', self.node_loop)
        self.register_node('parallel', self.node_parallel)
    
    def register_node(self, name: str, func):
        """注册任务节点"""
        self.nodes[name] = func
    
    async def execute_workflow(self, workflow: Dict, context: Dict = None) -> Dict:
        """执行工作流"""
        if context is None:
            context = {}
        
        workflow_id = f"wf_{datetime.now().timestamp()}"
        self.workflows[workflow_id] = {
            'status': 'running',
            'started_at': datetime.now(),
            'nodes': {}
        }
        
        try:
            # 解析工作流
            nodes = workflow['nodes']
            edges = workflow['edges']
            
            # 构建执行图
            graph = self._build_graph(nodes, edges)
            
            # 拓扑排序
            execution_order = self._topological_sort(graph)
            
            # 按顺序执行节点
            results = {}
            for node_id in execution_order:
                node = nodes[node_id]
                node_type = node['type']
                
                # 获取输入
                inputs = self._resolve_inputs(node, results, context)
                
                # 执行节点
                if node_type in self.nodes:
                    output = await self.nodes[node_type](inputs, context)
                    results[node_id] = output
                    self.workflows[workflow_id]['nodes'][node_id] = {
                        'status': 'completed',
                        'output': output
                    }
                else:
                    raise ValueError(f"未知节点类型：{node_type}")
            
            # 返回最终结果
            self.workflows[workflow_id]['status'] = 'completed'
            return {
                'workflow_id': workflow_id,
                'status': 'completed',
                'results': results
            }
            
        except Exception as e:
            self.workflows[workflow_id]['status'] = 'failed'
            self.workflows[workflow_id]['error'] = str(e)
            raise
    
    def _build_graph(self, nodes: Dict, edges: List) -> Dict:
        """构建执行图"""
        graph = {node_id: [] for node_id in nodes}
        for edge in edges:
            graph[edge['source']].append(edge['target'])
        return graph
    
    def _topological_sort(self, graph: Dict) -> List:
        """拓扑排序"""
        visited = set()
        result = []
        
        def visit(node):
            if node in visited:
                return
            visited.add(node)
            for neighbor in graph.get(node, []):
                visit(neighbor)
            result.append(node)
        
        for node in graph:
            visit(node)
        
        return result[::-1]  # 反转得到执行顺序
    
    def _resolve_inputs(self, node: Dict, results: Dict, context: Dict) -> Dict:
        """解析节点输入"""
        inputs = node.get('inputs', {})
        resolved = {}
        
        for key, value in inputs.items():
            if isinstance(value, str) and value.startswith('${'):
                # 引用其他节点的输出
                ref = value[2:-1]  # 去掉 ${ }
                if '.' in ref:
                    node_id, output_key = ref.split('.')
                    resolved[key] = results.get(node_id, {}).get(output_key)
                else:
                    resolved[key] = context.get(ref)
            else:
                resolved[key] = value
        
        return resolved
    
    # ========== 内置节点实现 ==========
    
    async def node_read_file(self, inputs: Dict, context: Dict) -> Dict:
        """读取文件节点"""
        file_path = inputs.get('path')
        encoding = inputs.get('encoding', 'utf-8')
        
        with open(file_path, 'r', encoding=encoding) as f:
            content = f.read()
        
        return {'content': content, 'size': len(content)}
    
    async def node_write_file(self, inputs: Dict, context: Dict) -> Dict:
        """写入文件节点"""
        file_path = inputs.get('path')
        content = inputs.get('content')
        encoding = inputs.get('encoding', 'utf-8')
        
        with open(file_path, 'w', encoding=encoding) as f:
            f.write(content)
        
        return {'success': True, 'path': file_path}
    
    async def node_list_files(self, inputs: Dict, context: Dict) -> Dict:
        """列出文件节点"""
        directory = inputs.get('directory')
        pattern = inputs.get('pattern', '*')
        
        from pathlib import Path
        files = list(Path(directory).glob(pattern))
        
        return {
            'files': [str(f) for f in files],
            'count': len(files)
        }
    
    async def node_parse_csv(self, inputs: Dict, context: Dict) -> Dict:
        """解析 CSV 节点"""
        import csv
        from io import StringIO
        
        content = inputs.get('content')
        reader = csv.DictReader(StringIO(content))
        rows = list(reader)
        
        return {'data': rows, 'rows': len(rows)}
    
    async def node_ai_analyze(self, inputs: Dict, context: Dict) -> Dict:
        """AI 分析节点"""
        content = inputs.get('content')
        prompt = inputs.get('prompt', '分析以下内容')
        
        # 调用本地 AI 模型
        from ...api.inference import get_model
        model = get_model(context.get('model', 'default'))
        
        response = model.generate([
            {'role': 'user', 'content': f'{prompt}:\n\n{content}'}
        ])
        
        return {'analysis': response}
    
    async def node_condition(self, inputs: Dict, context: Dict) -> Dict:
        """条件判断节点"""
        condition = inputs.get('condition')
        true_branch = inputs.get('true_branch')
        false_branch = inputs.get('false_branch')
        
        # 评估条件
        result = eval(condition, {'__builtins__': {}}, context)
        
        return {
            'result': result,
            'branch': true_branch if result else false_branch
        }
    
    async def node_parallel(self, inputs: Dict, context: Dict) -> Dict:
        """并行执行节点"""
        tasks = inputs.get('tasks', [])
        results = await asyncio.gather(*[
            self.execute_workflow(task, context) for task in tasks
        ])
        
        return {'results': results}
```

```python
# server/privacy/secure_erase.py

import os
import shutil
from pathlib import Path
import secrets

class SecureEraser:
    """安全擦除器 - 彻底删除敏感数据"""
    
    def __init__(self, data_dir: str):
        self.data_dir = Path(data_dir)
    
    def secure_delete(self, file_path: str, passes: int = 3):
        """安全删除文件（多次覆盖）"""
        path = Path(file_path)
        
        if not path.exists():
            return
        
        # 获取文件大小
        file_size = path.stat().st_size
        
        # 多次覆盖写入
        for _ in range(passes):
            with open(path, 'wb') as f:
                # 写入随机数据
                f.write(secrets.token_bytes(file_size))
                f.flush()
                os.fsync(f.fileno())
        
        # 删除文件
        path.unlink()
    
    def clear_all_conversations(self):
        """清除所有对话记录"""
        conv_dir = self.data_dir / 'conversations'
        if conv_dir.exists():
            for file in conv_dir.glob('*.json'):
                self.secure_delete(str(file))
    
    def clear_memories(self, keep_importance: float = 0.0):
        """清除记忆（可保留高重要度记忆）"""
        from ..memory.database import MemoryDatabase
        
        db = MemoryDatabase(str(self.data_dir))
        
        # 获取所有记忆
        memories = db.get_memories_by_importance(min_importance=0.0)
        
        for memory in memories:
            if memory['importance'] < keep_importance:
                db.forget_memory(memory['id'])
    
    def clear_cache(self):
        """清除缓存"""
        cache_dir = self.data_dir / 'cache'
        if cache_dir.exists():
            shutil.rmtree(cache_dir)
    
    def clear_logs(self):
        """清除日志"""
        log_dir = self.data_dir / 'logs'
        if log_dir.exists():
            for file in log_dir.glob('*.log'):
                self.secure_delete(str(file))
    
    def clear_models(self):
        """清除下载的模型"""
        models_dir = self.data_dir / 'models'
        if models_dir.exists():
            shutil.rmtree(models_dir)
    
    def one_click_privacy(self):
        """一键隐私模式 - 清除所有痕迹"""
        import asyncio
        
        # 并行执行所有清理任务
        tasks = [
            asyncio.create_task(self.clear_all_conversations()),
            asyncio.create_task(self.clear_memories()),
            asyncio.create_task(self.clear_cache()),
            asyncio.create_task(self.clear_logs())
        ]
        
        await asyncio.gather(*tasks)
        
        return {
            'status': 'completed',
            'cleared': ['conversations', 'memories', 'cache', 'logs']
        }
```
