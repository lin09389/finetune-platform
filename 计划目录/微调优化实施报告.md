### P0-1: 检查点保存异步化

**问题**: 检查点保存阻塞主线程，训练卡顿 2-5 秒

**实施方案**:
```python
class ProgressCallback:
    def __init__(self, ...):
        # 后台线程池执行保存
        self._checkpoint_executor = ThreadPoolExecutor(
            max_workers=1, 
            thread_name_prefix="checkpoint_saver"
        )
    
    def _save_checkpoint_async(self):
        """异步保存检查点 - 后台线程执行，不阻塞训练"""
        future = self._checkpoint_executor.submit(self._do_save_checkpoint)
        self._pending_checkpoint = future
    
    def on_train_end(self, ...):
        # 等待最后的检查点保存完成
        if self._pending_checkpoint:
            self._pending_checkpoint.result(timeout=300)
```

**修改文件**: `server/api/training.py` (行 287-465)

**预期收益**:
- ✅ 训练过程无卡顿

```python
def cleanup_gpu_memory(aggressive: bool = False):
    """增强版显存清理"""
    # 1-4: 基础清理
    torch.cuda.synchronize()
    torch.cuda.empty_cache()
    torch.cuda.reset_peak_memory_stats()
    gc.collect()
    
    # 5: 激进模式 - 深度清理所有 GPU 设备
    if aggressive:
        for i in range(torch.cuda.device_count()):
            with torch.cuda.device(i):
                # 多次清理确保彻底
                for _ in range(3):
                    torch.cuda.empty_cache()
                    torch.cuda.synchronize()
    
    # 6-7: 记录清理后状态
    vram_used = torch.cuda.memory_allocated(0) / (1024 ** 3)
    vram_total = torch.cuda.get_device_properties(0).total_memory / (1024 ** 3)
    logger.info(f"显存清理完成：{vram_used:.2f}/{vram_total:.2f}GB")
```

### P0-3: 进度更新频率优化

**问题**: 每步更新进度导致 CPU 占用高，前端渲染跟不上

**实施方案**:
```python
class ProgressCallback:
    def __init__(self, ...):
        # 降低更新频率
        self.last_update_step = -1
        self.update_interval = max(1, config.logging_steps)
    
    def on_step_end(self, args, state, control, **kwargs):
        # 每 N 步更新一次
        if (self.current_step - self.last_update_step) >= self.update_interval:
            self._update_progress(state, args, kwargs)
            self.last_update_step = self.current_step
```

### P1-1: 异常恢复机制

**问题**: 训练中断后需手动恢复，无法自动处理临时错误

**实施方案**:

**1. 异常分类**:

```python
class RecoverableError(Exception):
    """可恢复错误 - OOM、GPU 错误等"""
    pass

class UnrecoverableError(Exception):
    """不可恢复错误 - 文件丢失、用户中断等"""
    pass
```

**2. 异常捕获**:
```python
def training_thread(config, ..., retry_count=0):
    try:
        # 加载模型
        try:
            model, tokenizer = load_model_and_tokenizer(...)
        except torch.cuda.OutOfMemoryError:
            raise RecoverableError("加载模型时 OOM")
        except FileNotFoundError:
            raise UnrecoverableError("模型文件丢失")
        
        # 训练过程
        try:
            trainer.train()
        except torch.cuda.OutOfMemoryError:
            raise RecoverableError("训练时 OOM")
        except KeyboardInterrupt:
            raise UnrecoverableError("用户中断")
    
    except RecoverableError as e:
        # 自动重试 + 智能降级
        if retry_count < max_retries:
            degraded_config = _degrade_training_config(config)
            return training_thread(degraded_config, ..., retry_count + 1)
```

**3. 智能降级配置**:
```python
def _degrade_training_config(config):
    """根据显存情况自动调整参数"""
    degraded = config.model_copy()
    
    free_vram = available_vram - allocated_vram
    
    if free_vram < 4.0:
        # 激进降级
        degraded.batch_size = 1
        degraded.gradient_accumulation = 8
        degraded.max_seq_length = 256
    elif free_vram < 6.0:
        # 保守降级
        degraded.batch_size = 2
        degraded.gradient_accumulation = 16
    
    return degraded
```

```python
class TrainingLogger:
    """训练日志记录器"""
    
    def __init__(self, task_id, output_dir):
        self.task_id = task_id
        self.log_file = output_dir / "training.log"
        self.metrics_file = output_dir / "metrics.jsonl"
        self.events_file = output_dir / "events.jsonl"
    
    def log_start(self, config):
        """记录训练开始"""
        self.logger.info("=" * 60)
        self.logger.info("训练开始")
        self.logger.info(f"任务 ID: {self.task_id}")
        self.logger.info(f"模型：{config.model_id}")
        # ... 记录所有配置
    
    def log_metrics(self, epoch, step, metrics):
        """记录训练指标"""
        metrics_record = {
            "timestamp": datetime.now().isoformat(),
            "epoch": epoch,
            "step": step,
            **metrics
        }
        with open(self.metrics_file, 'a') as f:
            f.write(json.dumps(metrics_record) + '\n')
    
    def log_event(self, event_type, data):
        """记录训练事件"""
        self._log_event(event_type, data)
    
    def log_error(self, error, context=None):
        """记录错误"""
        self.logger.error(f"错误：{error}", exc_info=True)
        self._log_event("error", {...})
    
    def log_checkpoint_saved(self, step, path):
        """记录检查点保存"""
        self.logger.info(f"检查点保存：step={step}, path={path}")
    
    def log_completion(self, final_metrics):
        """记录训练完成"""
        self.logger.info("=" * 60)
        self.logger.info("训练完成")
        self.logger.info(f"最终 Loss: {final_metrics.get('loss')}")
```

**输出文件**:
- `training.log` - 结构化日志
- `metrics.jsonl` - 训练指标（每步）
- `events.jsonl` - 训练事件
