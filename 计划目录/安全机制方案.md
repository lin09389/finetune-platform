### 第 1 步：API Key 加密存储

```python
# server/security/encryption.py

from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
import base64
import os
import json
from pathlib import Path

class SecureStorage:
    """安全存储 - 加密敏感数据"""
    
    def __init__(self, master_password: str = None):
        """
        初始化加密存储
        
        Args:
            master_password: 主密码（可选，不提供则使用系统密钥）
        """
        self.key_file = Path(__file__).parent.parent / "data" / ".encryption_key"
        self.vault_file = Path(__file__).parent.parent / "data" / ".vault"
        
        # 获取或生成加密密钥
        self.key = self._get_or_create_key(master_password)
        self.cipher = Fernet(self.key)
    
    def _get_or_create_key(self, master_password: str = None) -> bytes:
        """获取或创建加密密钥"""
        if self.key_file.exists():
            # 读取现有密钥
            with open(self.key_file, 'rb') as f:
                encrypted_key = f.read()
            
            if master_password:
                # 用主密码解密
                kdf = PBKDF2HMAC(
                    algorithm=hashes.SHA256(),
                    length=32,
                    salt=b'static_salt_change_this',  # 应该用随机 salt
                    iterations=100000,
                )
                key = base64.urlsafe_b64encode(kdf.derive(master_password.encode()))
                cipher = Fernet(key)
                return cipher.decrypt(encrypted_key)
            else:
                return encrypted_key
        else:
            # 生成新密钥
            key = Fernet.generate_key()
            
            if master_password:
                # 用主密码加密后存储
                kdf = PBKDF2HMAC(
                    algorithm=hashes.SHA256(),
                    length=32,
                    salt=b'static_salt_change_this',
                    iterations=100000,
                )
                encrypt_key = base64.urlsafe_b64encode(kdf.derive(master_password.encode()))
                cipher = Fernet(encrypt_key)
                encrypted_key = cipher.encrypt(key)
            else:
                encrypted_key = key
            
            # 保存密钥
            self.key_file.parent.mkdir(parents=True, exist_ok=True)
            with open(self.key_file, 'wb') as f:
                f.write(encrypted_key)
            
            # 设置安全权限（仅所有者可读写）
            os.chmod(self.key_file, 0o600)
            
            return key
    
    def encrypt(self, plaintext: str) -> str:
        """加密字符串"""
        encrypted = self.cipher.encrypt(plaintext.encode())
        return base64.urlsafe_b64encode(encrypted).decode()
    
    def decrypt(self, ciphertext: str) -> str:
        """解密字符串"""
        encrypted = base64.urlsafe_b64decode(ciphertext.encode())
        decrypted = self.cipher.decrypt(encrypted)
        return decrypted.decode()
    
    def store_secret(self, key: str, value: str):
        """存储密钥值对"""
        vault = self._load_vault()
        vault[key] = self.encrypt(value)
        self._save_vault(vault)
    
    def get_secret(self, key: str, default: str = None) -> str:
        """获取密钥值"""
        vault = self._load_vault()
        encrypted = vault.get(key)
        if not encrypted:
            return default
        return self.decrypt(encrypted)
    
    def delete_secret(self, key: str):
        """删除密钥"""
        vault = self._load_vault()
        if key in vault:
            del vault[key]
            self._save_vault(vault)
    
    def _load_vault(self) -> dict:
        """加载保险库"""
        if self.vault_file.exists():
            with open(self.vault_file, 'r') as f:
                return json.load(f)
        return {}
    
    def _save_vault(self, vault: dict):
        """保存保险库"""
        with open(self.vault_file, 'w') as f:
            json.dump(vault, f, indent=2)
        # 设置安全权限
        os.chmod(self.vault_file, 0o600)


# 全局实例
secure_storage = SecureStorage()
```

```python
# server/api/api_keys.py (安全增强版)

from security.encryption import secure_storage
import hashlib
import secrets

class APIKeyManager:
    """API Key 管理器 - 安全版本"""
    
    def __init__(self):
        self.db_path = Path(__file__).parent.parent / "data" / "api_keys.db"
        self._init_db()
    
    def _init_db(self):
        """初始化数据库"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS api_keys (
                id TEXT PRIMARY KEY,
                provider TEXT NOT NULL,
                key_hash TEXT NOT NULL,
                key_encrypted TEXT NOT NULL,  -- 加密存储
                key_prefix TEXT,
                name TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                last_used TIMESTAMP,
                usage_count INTEGER DEFAULT 0,
                is_active INTEGER DEFAULT 1,
                daily_limit INTEGER DEFAULT -1,  -- -1 表示无限制
                monthly_usage REAL DEFAULT 0
            )
        """)
        
        conn.commit()
        conn.close()
    
    def add_key(
        self,
        provider: str,
        api_key: str,
        name: str = None,
        daily_limit: int = -1
    ) -> str:
        """添加 API Key（加密存储）"""
        import uuid
        
        key_id = f"key_{uuid.uuid4().hex[:8]}"
        
        # 哈希（用于快速查找和验证）
        key_hash = hashlib.sha256(api_key.encode()).hexdigest()
        
        # 加密存储
        key_encrypted = secure_storage.encrypt(api_key)
        
        # 前缀（用于显示）
        key_prefix = f"{api_key[:8]}..." if len(api_key) > 8 else api_key
        
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            INSERT INTO api_keys 
            (id, provider, key_hash, key_encrypted, key_prefix, name, daily_limit)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (key_id, provider, key_hash, key_encrypted, key_prefix, name or provider, daily_limit))
        
        conn.commit()
        conn.close()
        
        return key_id
    
    def get_key(self, key_id: str) -> dict:
        """获取 API Key（解密后返回）"""
        conn = sqlite3.connect(self.db_path)
        conn.row_factory = sqlite3.Row
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT * FROM api_keys WHERE id = ? AND is_active = 1
        """, (key_id,))
        
        row = cursor.fetchone()
        conn.close()
        
        if not row:
            return None
        
        # 解密 API Key
        try:
            api_key = secure_storage.decrypt(row['key_encrypted'])
        except:
            # 解密失败（可能密钥丢失）
            return None
        
        return {
            'id': row['id'],
            'provider': row['provider'],
            'api_key': api_key,
            'name': row['name'],
            'daily_limit': row['daily_limit'],
            'monthly_usage': row['monthly_usage']
        }
    
    def get_key_info(self, key_id: str) -> dict:
        """获取 API Key 信息（不解密，用于显示）"""
        conn = sqlite3.connect(self.db_path)
        conn.row_factory = sqlite3.Row
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT id, provider, key_prefix, name, created_at, last_used, 
                   usage_count, is_active, daily_limit
            FROM api_keys WHERE id = ?
        """, (key_id,))
        
        row = cursor.fetchone()
        conn.close()
        
        if not row:
            return None
        
        return dict(row)
    
    def delete_key(self, key_id: str):
        """删除 API Key（同时删除加密数据）"""
        # 先删除保险库中的加密数据
        secure_storage.delete_secret(f"api_key:{key_id}")
        
        # 再删除数据库记录
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("DELETE FROM api_keys WHERE id = ?", (key_id,))
        
        conn.commit()
        conn.close()
    
    def check_rate_limit(self, key_id: str) -> bool:
        """检查速率限制"""
        key_info = self.get_key(key_id)
        if not key_info:
            return False
        
        daily_limit = key_info.get('daily_limit', -1)
        if daily_limit < 0:  # 无限制
            return True
        
        # 检查今日使用量
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT usage_count FROM api_keys 
            WHERE id = ? AND date(last_used) = date('now')
        """, (key_id,))
        
        row = cursor.fetchone()
        today_usage = row[0] if row else 0
        
        conn.close()
        
        return today_usage < daily_limit
    
    def record_usage(self, key_id: str, tokens: int = 1):
        """记录使用量"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("""
            UPDATE api_keys 
            SET last_used = CURRENT_TIMESTAMP, 
                usage_count = usage_count + 1,
                monthly_usage = monthly_usage + ?
            WHERE id = ?
        """, (tokens, key_id))
        
        conn.commit()
        conn.close()
```

```python
# server/security/file_sandbox.py

from pathlib import Path
import os
import re

class FileSandbox:
    """文件操作沙箱 - 限制文件访问范围"""
    
    # 禁止访问的路径模式
    FORBIDDEN_PATTERNS = [
        r'\.\.',  # 目录遍历
        r'^/etc/',
        r'^/usr/',
        r'^/var/',
        r'^/root/',
        r'^C:/Windows/',
        r'^C:/Program Files/',
        r'^/Users/.*/Library/',
    ]
    
    # 允许的操作
    ALLOWED_OPERATIONS = [
        'read', 'write', 'create', 'delete', 'list', 'move', 'copy'
    ]
    
    # 危险文件扩展名
    DANGEROUS_EXTENSIONS = [
        '.exe', '.bat', '.cmd', '.sh', '.bash',
        '.dll', '.so', '.dylib',
        '.ps1', '.vbs', '.js', '.jar'
    ]
    
    def __init__(self, working_dir: str = None):
        """
        初始化沙箱
        
        Args:
            working_dir: 工作目录（默认项目根目录）
        """
        if working_dir:
            self.working_dir = Path(working_dir).resolve()
        else:
            self.working_dir = Path(__file__).parent.parent.parent.resolve()
    
    def safe_path(self, file_path: str, operation: str = 'read') -> Path:
        """
        安全检查文件路径
        
        Args:
            file_path: 文件路径
            operation: 操作类型
        
        Returns:
            安全的路径对象
        
        Raises:
            PermissionError: 访问被拒绝
            ValueError: 路径无效
        """
        # 检查操作是否允许
        if operation not in self.ALLOWED_OPERATIONS:
            raise PermissionError(f"不允许的操作：{operation}")
        
        # 转换为绝对路径
        if os.path.isabs(file_path):
            path = Path(file_path).resolve()
        else:
            path = (self.working_dir / file_path).resolve()
        
        # 检查是否在工作目录内
        try:
            path.relative_to(self.working_dir)
        except ValueError:
            raise PermissionError(
                f"禁止访问工作目录外的文件：{file_path}\n"
                f"工作目录：{self.working_dir}"
            )
        
        # 检查禁止模式
        path_str = str(path).replace('\\', '/')
        for pattern in self.FORBIDDEN_PATTERNS:
            if re.search(pattern, path_str, re.IGNORECASE):
                raise PermissionError(f"禁止访问的路径：{file_path}")
        
        # 检查危险文件
        if operation in ['write', 'create']:
            ext = path.suffix.lower()
            if ext in self.DANGEROUS_EXTENSIONS:
                raise PermissionError(
                    f"禁止创建危险文件类型：{ext}\n"
                    f"如需创建可执行文件，请手动操作"
                )
        
        return path
    
    def read_file(self, file_path: str, max_size: int = 10*1024*1024) -> str:
        """安全读取文件"""
        path = self.safe_path(file_path, 'read')
        
        # 检查文件大小
        if path.exists():
            file_size = path.stat().st_size
            if file_size > max_size:
                raise PermissionError(
                    f"文件过大：{file_size/1024/1024:.2f}MB\n"
                    f"最大允许：{max_size/1024/1024:.2f}MB"
                )
        
        with open(path, 'r', encoding='utf-8', errors='ignore') as f:
            return f.read()
    
    def write_file(self, file_path: str, content: str) -> str:
        """安全写入文件"""
        path = self.safe_path(file_path, 'write')
        
        with open(path, 'w', encoding='utf-8') as f:
            f.write(content)
        
        return str(path)
    
    def create_file(self, file_path: str, content: str = "") -> str:
        """安全创建文件"""
        path = self.safe_path(file_path, 'create')
        
        # 创建父目录
        path.parent.mkdir(parents=True, exist_ok=True)
        
        with open(path, 'w', encoding='utf-8') as f:
            f.write(content)
        
        return str(path)
    
    def delete_file(self, file_path: str) -> bool:
        """安全删除文件"""
        path = self.safe_path(file_path, 'delete')
        
        # 禁止删除目录
        if path.is_dir():
            raise PermissionError("禁止删除目录")
        
        path.unlink()
        return True
    
    def list_files(self, directory: str = ".", pattern: str = "*") -> list:
        """安全列出文件"""
        dir_path = self.safe_path(directory, 'list')
        
        if not dir_path.is_dir():
            raise ValueError(f"不是目录：{directory}")
        
        files = []
        for item in dir_path.glob(pattern):
            # 跳过隐藏文件
            if item.name.startswith('.'):
                continue
            
            files.append({
                'name': item.name,
                'path': str(item.relative_to(self.working_dir)),
                'is_dir': item.is_dir(),
                'size': item.stat().st_size if item.is_file() else 0
            })
        
        return files
    
    def get_allowed_operations(self) -> list:
        """获取允许的操作列表"""
        return self.ALLOWED_OPERATIONS.copy()
    
    def get_working_dir(self) -> str:
        """获取工作目录"""
        return str(self.working_dir)
```

```python
# server/security/input_validation.py

import re
from typing import Any, Dict, List
from pydantic import BaseModel, validator, Field

class InputValidator:
    """输入验证器"""
    
    # 最大输入长度
    MAX_INPUT_LENGTH = 100000
    MAX_FILE_PATH_LENGTH = 500
    MAX_MESSAGE_LENGTH = 10000
    
    # 危险字符模式
    DANGEROUS_PATTERNS = [
        r'<script[^>]*>',  # XSS
        r'javascript:',    # JS 协议
        r'data:',          # Data URI
        r'{{.*}}',         # 模板注入
        r'\$\{.*\}',       # 表达式注入
        r'__import__',     # Python 注入
        r'eval\s*\(',      # eval 调用
        r'exec\s*\(',      # exec 调用
    ]
    
    @classmethod
    def validate_string(cls, value: str, max_length: int = None) -> str:
        """验证字符串"""
        if not isinstance(value, str):
            raise ValueError("必须是字符串")
        
        if max_length is None:
            max_length = cls.MAX_INPUT_LENGTH
        
        if len(value) > max_length:
            raise ValueError(f"输入过长：{len(value)}/{max_length}")
        
        # 检查危险模式
        for pattern in cls.DANGEROUS_PATTERNS:
            if re.search(pattern, value, re.IGNORECASE):
                raise ValueError(f"检测到危险内容：{pattern}")
        
        return value.strip()
    
    @classmethod
    def validate_file_path(cls, path: str) -> str:
        """验证文件路径"""
        if not isinstance(path, str):
            raise ValueError("路径必须是字符串")
        
        if len(path) > cls.MAX_FILE_PATH_LENGTH:
            raise ValueError("路径过长")
        
        # 检查危险字符
        if '..' in path:
            raise ValueError("禁止目录遍历")
        
        if path.startswith('/') or path.startswith('\\'):
            raise ValueError("禁止绝对路径")
        
        return path.strip()
    
    @classmethod
    def validate_chat_message(cls, messages: List[Dict]) -> List[Dict]:
        """验证聊天消息"""
        if not isinstance(messages, list):
            raise ValueError("消息必须是列表")
        
        if len(messages) > 100:
            raise ValueError("消息数量过多")
        
        validated = []
        for msg in messages:
            if not isinstance(msg, dict):
                raise ValueError("消息必须是字典")
            
            if 'role' not in msg or 'content' not in msg:
                raise ValueError("消息必须包含 role 和 content")
            
            if msg['role'] not in ['user', 'assistant', 'system']:
                raise ValueError("无效的 role")
            
            content = cls.validate_string(
                msg['content'],
                cls.MAX_MESSAGE_LENGTH
            )
            
            validated.append({
                'role': msg['role'],
                'content': content
            })
        
        return validated
    
    @classmethod
    def sanitize_html(cls, text: str) -> str:
        """清理 HTML（防止 XSS）"""
        # 替换危险字符
        text = text.replace('&', '&amp;')
        text = text.replace('<', '&lt;')
        text = text.replace('>', '&gt;')
        text = text.replace('"', '&quot;')
        text = text.replace("'", '&#x27;')
        return text
    
    @classmethod
    def validate_api_key_format(cls, api_key: str, provider: str) -> bool:
        """验证 API Key 格式"""
        if not api_key or len(api_key) < 10:
            return False
        
        if provider == 'minimax':
            # Minimax 格式：group_id:api_key
            if ':' not in api_key:
                return False
            parts = api_key.split(':')
            if len(parts) != 2:
                return False
            if not parts[0].isdigit():
                return False
        
        elif provider == 'glm':
            # 智谱格式：xxxxx.yyyyy
            if '.' not in api_key:
                return False
        
        return True


# Pydantic 模型示例
class SafeChatRequest(BaseModel):
    """安全的聊天请求"""
    
    messages: List[Dict[str, str]]
    model: str = Field(..., max_length=100)
    temperature: float = Field(default=0.7, ge=0.0, le=2.0)
    max_tokens: int = Field(default=1024, ge=1, le=10000)
    
    @validator('messages')
    def validate_messages(cls, v):

    @validator('model')
    def validate_model(cls, v):

    content: str = ""
    
    @validator('file_path')
    def validate_file_path(cls, v):
        return InputValidator.validate_file_path(v)
    
    @validator('content')
    def validate_content(cls, v):
        return InputValidator.validate_string(v)
```

```python
# server/security/audit_log.py

import logging
import json
from datetime import datetime
from pathlib import Path
from typing import Dict, Any

class AuditLogger:
    """审计日志 - 记录所有敏感操作"""
    
    def __init__(self, log_dir: str = None):
        if log_dir:
            self.log_dir = Path(log_dir)
        else:
            self.log_dir = Path(__file__).parent.parent / "logs" / "audit"
        
        self.log_dir.mkdir(parents=True, exist_ok=True)
        
        # 配置日志
        self.logger = logging.getLogger('audit')
        self.logger.setLevel(logging.INFO)
        
        # 文件处理器
        log_file = self.log_dir / f"audit_{datetime.now().strftime('%Y%m%d')}.log"
        handler = logging.FileHandler(log_file, encoding='utf-8')
        handler.setFormatter(logging.Formatter(
            '%(asctime)s - %(levelname)s - %(message)s'
        ))
        self.logger.addHandler(handler)
        
        # 设置权限（仅所有者可读写）
        try:
            import os
            os.chmod(log_file, 0o600)
        except:
            pass
    
    def log_action(
        self,
        action: str,
        user_id: str = "anonymous",
        resource: str = None,
        details: Dict[str, Any] = None,
        success: bool = True,
        ip_address: str = None
    ):
        """记录操作日志"""
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'action': action,
            'user_id': user_id,
            'resource': resource,
            'details': details or {},
            'success': success,
            'ip_address': ip_address
        }
        
        log_level = logging.INFO if success else logging.WARNING
        
        self.logger.log(
            log_level,
            json.dumps(log_entry, ensure_ascii=False)
        )
    
    def log_api_key_access(self, key_id: str, provider: str, success: bool):
        """记录 API Key 使用"""
        self.log_action(
            action='api_key_access',
            resource=key_id,
            details={'provider': provider},
            success=success
        )
    
    def log_file_operation(
        self,
        operation: str,
        file_path: str,
        user_id: str = "anonymous",
        success: bool = True
    ):
        """记录文件操作"""
        self.log_action(
            action=f'file_{operation}',
            user_id=user_id,
            resource=file_path,
            success=success
        )
    
    def log_login_attempt(
        self,
        user_id: str,
        success: bool,
        ip_address: str = None,
        reason: str = None
    ):
        """记录登录尝试"""
        self.log_action(
            action='login_attempt',
            user_id=user_id,
            success=success,
            details={'reason': reason} if reason else {},
            ip_address=ip_address
        )
    
    def log_security_event(
        self,
        event_type: str,
        details: Dict[str, Any],
        severity: str = 'medium'
    ):
        """记录安全事件"""
        self.log_action(
            action=f'security_{event_type}',
            details={**details, 'severity': severity},
            success=(severity != 'critical')
        )
    
    def get_logs(
        self,
        date: str = None,
        action_filter: str = None,
        limit: int = 100
    ) -> list:
        """获取日志（用于管理界面）"""
        if not date:
            date = datetime.now().strftime('%Y%m%d')
        
        log_file = self.log_dir / f"audit_{date}.log"
        
        if not log_file.exists():
            return []
        
        logs = []
        with open(log_file, 'r', encoding='utf-8') as f:
            for line in f:
                try:
                    # 解析日志
                    parts = line.split(' - ', 3)
                    if len(parts) >= 4:
                        log_json = parts[3]
                        log_entry = json.loads(log_json)
                        
                        # 过滤
                        if action_filter and log_entry.get('action') != action_filter:
                            continue
                        
                        logs.append(log_entry)
                        
                        if len(logs) >= limit:
                            break
                except:
                    continue
        
        return logs
```

```python
# server/security/middleware.py

from fastapi import Request, HTTPException, status
from fastapi.responses import JSONResponse
import time
from collections import defaultdict
from datetime import datetime, timedelta

class SecurityMiddleware:
    """安全中间件"""
    
    def __init__(self):
        # 速率限制存储
        self.rate_limits = defaultdict(list)
        
        # IP 黑名单
        self.blacklist = set()
        
        # 请求计数
        self.request_counts = defaultdict(int)
    
    async def __call__(self, request: Request, call_next):
        """中间件调用"""
        client_ip = request.client.host
        path = request.url.path
        
        # 检查黑名单
        if client_ip in self.blacklist:
            return JSONResponse(
                status_code=403,
                content={"error": "IP 已被封禁"}
            )
        
        # 速率限制检查
        if not self.check_rate_limit(client_ip):
            from security.audit_log import audit_logger
            audit_logger.log_security_event(
                'rate_limit_exceeded',
                {'ip': client_ip, 'path': path},
                'medium'
            )
            return JSONResponse(
                status_code=429,
                content={"error": "请求过于频繁"}
            )
        
        # 记录请求
        start_time = time.time()
        
        try:
            response = await call_next(request)
            
            # 记录成功请求
            duration = time.time() - start_time
            self.log_request(client_ip, path, duration, True)
            
            return response
            
        except Exception as e:
            # 记录失败请求
            duration = time.time() - start_time
            self.log_request(client_ip, path, duration, False)
            
            # 记录异常
            from security.audit_log import audit_logger
            audit_logger.log_security_event(
                'request_error',
                {'ip': client_ip, 'path': path, 'error': str(e)},
                'low'
            )
            
            raise
    
    def check_rate_limit(self, client_ip: str, limit: int = 100, window: int = 60) -> bool:
        """检查速率限制"""
        now = time.time()
        window_start = now - window
        
        # 清理过期记录
        self.rate_limits[client_ip] = [
            ts for ts in self.rate_limits[client_ip] if ts > window_start
        ]
        
        # 检查是否超限
        if len(self.rate_limits[client_ip]) >= limit:
            return False
        
        # 记录当前请求
        self.rate_limits[client_ip].append(now)
        return True
    
    def log_request(self, client_ip: str, path: str, duration: float, success: bool):
        """记录请求"""
        self.request_counts[client_ip] += 1
        
        # 异常检测（大量失败）
        if not success:
            if self.request_counts[client_ip] > 100:
                # 失败率过高，加入黑名单
                self.blacklist.add(client_ip)
                
                from security.audit_log import audit_logger
                audit_logger.log_security_event(
                    'ip_blacklisted',
                    {'ip': client_ip, 'reason': 'high_failure_rate'},
                    'high'
                )
```
